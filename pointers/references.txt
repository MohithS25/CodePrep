Q1. What is the difference between ref = y and p = &y when ref is a reference to x and p is a pointer to x?
ref=y updates the value at x and makes x = y, p = &y redirects the pointer p to point to the address of y and to access value at y we need to dereference p  

Q2. Why can't a reference be null, but a pointer can? What does this mean practically when writing functions?
reference always points to some variable and hence you cant create a reference to something that doesn't exist whereas pointers can be null because you can create them and not point to any address but can reassign it later to point to some address

#include <iostream>

void find(int* arr, int size, int key, bool& found, int& value){
    
    found = false;
    for(int i=0;i<size;i++){
        if(arr[i] == key){
            found = true;
            value = arr[i];

    }

Exercise 2 — Min and Max
  Write a function that finds both the minimum and maximum of an array in a single pass. Return both through reference parameters.

#include <iostream>

void min_max(int* arr,int size,int& min,int& max){
    min = max = arr[0];
    for(int i=0;i<size;i++){
        if(size == 1){min = max = arr[i];}
        if(arr[i]<min){
            min = arr[i];
        }
        else if(arr[i]>max){
            max = arr[i];
        }
    }
}

int main(){
    int nums[] = {3,7,1,9,2,-8,4};
    int min,max;
    min_max(nums,7,min,max);
    std::cout<<"Max: "<<max<<"\nMin: "<<min<<std::endl;
    return 0;
}
